cmake_minimum_required(VERSION 3.25)
project(apsis_v2 CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(apsis_proto src/proto/proto.cpp)
target_include_directories(apsis_proto PUBLIC src)
add_executable(gates tools/gates.cpp)
target_link_libraries(gates apsis_proto)
add_executable(bench tools/bench.cpp)
target_link_libraries(bench apsis_proto)
add_executable(params_md tools/params_md.cpp)
add_library(apsis_world src/world/serialize.cpp src/world/gen.cpp src/world/econ.cpp src/world/power.cpp src/world/mapdark.cpp src/world/sim.cpp src/world/intents.cpp)
target_include_directories(apsis_world PUBLIC src)
add_executable(world_port tools/world_port.cpp)
target_link_libraries(world_port apsis_world)
add_executable(world_gates tools/world_gates.cpp)
target_link_libraries(world_gates apsis_world apsis_proto)
add_executable(world_core tools/world_core.cpp)
target_link_libraries(world_core apsis_world)
add_executable(world_wide tools/world_wide.cpp)
target_link_libraries(world_wide apsis_world)
add_library(apsis_pilot src/pilot/pilot.cpp src/pilot/law.cpp src/pilot/bots.cpp)
target_include_directories(apsis_pilot PUBLIC src)
target_link_libraries(apsis_pilot apsis_world)
add_executable(pilot_exp tools/pilot_exp.cpp)
target_link_libraries(pilot_exp apsis_pilot)
add_executable(pilot_gates tools/pilot_gates.cpp)
target_link_libraries(pilot_gates apsis_pilot)
add_executable(pilot_bench tools/pilot_bench.cpp)
target_link_libraries(pilot_bench apsis_pilot)
